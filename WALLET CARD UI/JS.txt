// Restricts input for the given textbox to the given inputFilter.
function setInputFilter(textbox, inputFilter) {
  ["input", "keydown", "keyup", "mousedown", "mouseup", "select", "contextmenu", "drop"].forEach(function(event) {
    textbox.addEventListener(event, function() {
      if (inputFilter(this.value)) {
        this.oldValue = this.value;
        this.oldSelectionStart = this.selectionStart;
        this.oldSelectionEnd = this.selectionEnd;
      } else if (this.hasOwnProperty("oldValue")) {
        this.value = this.oldValue;
        this.setSelectionRange(this.oldSelectionStart, this.oldSelectionEnd);
      } else {
        this.value = "";
      }
    });
  });
}


// Install input filters.

  return /^\d*$/.test(value); });

  return /^[a-zA-Z ]*$/.test(value); });

	card.classList.remove("card_flipped");
	card_number_in_card.classList.add("focused");

	card_number_in_card.classList.remove("focused");
}


let card_number_value_sec=document.getElementsByClassName("card_number_value_sec");

card_number_in_form.addEventListener("input",function(event){

	card_number_value=event.target.value;
	
	let no_of_X=16-card_number_value.length;
	card_number_value=card_number_value.concat( "X".repeat(no_of_X) );
	
	// console.log(card_number_value);
	
	for(i=0;i<4;i=i+=1){
		card_number_value_sec[i].innerHTML=card_number_value.substring(i*4,i*4+4);	
	}
	
});
