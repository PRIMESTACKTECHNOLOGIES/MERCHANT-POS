package com.customer2013.data.api

import retrofit2.Response
import retrofit2.Retrofit
import retrofit2.converter.gson.GsonConverterFactory
import retrofit2.http.POST
import retrofit2.http.Query

data class GenerateResponse(
    val code: String,
    val amount: Double,
    val reference: String,
    val expiresAt: String? // "createdAt" from backend, essentially
)

interface PaymentApi {
    @POST("api/payment2013/generate")
    suspend fun generate(@Query("amount") amount: Double): Response<GenerateResponse>
}

object ApiClient {
    // IMPORTANT: Update this IP to your PC's IP address
    fun create(baseUrl: String = "http://192.168.1.160:5000/"): PaymentApi {
        return Retrofit.Builder()
            .baseUrl(baseUrl)
            .addConverterFactory(GsonConverterFactory.create())
            .build()
            .create(PaymentApi::class.java)
    }
}

package com.customer2013.ui

import android.os.Bundle
import android.view.View
import android.widget.Button
import android.widget.EditText
import android.widget.LinearLayout
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import com.customer2013.data.api.ApiClient
import com.customer2013.data.api.PaymentApi
import kotlinx.coroutines.launch
import com.customer2013.R // Placeholder

class MainActivity : AppCompatActivity() {

    private lateinit var api: PaymentApi

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        api = ApiClient.create()

        val edtAmount = findViewById<EditText>(R.id.edtAmount)
        val btnGenerate = findViewById<Button>(R.id.btnGenerate)
        val layoutResult = findViewById<LinearLayout>(R.id.layoutResult)
        val txtCode = findViewById<TextView>(R.id.txtCode)
        val txtDetails = findViewById<TextView>(R.id.txtDetails)
        val txtError = findViewById<TextView>(R.id.txtError)

        btnGenerate.setOnClickListener {
            val amountText = edtAmount.text.toString().trim()
            val amount = amountText.toDoubleOrNull()

            if (amount == null || amount <= 0) {
                txtError.text = "Please enter a valid amount"
                txtError.visibility = View.VISIBLE
                layoutResult.visibility = View.GONE
                return@setOnClickListener
            }

            // UI Loading state
            btnGenerate.isEnabled = false
            btnGenerate.text = "Generating..."
            txtError.visibility = View.GONE
            layoutResult.visibility = View.GONE

            lifecycleScope.launch {
                try {
                    val resp = api.generate(amount)

                    if (resp.isSuccessful && resp.body() != null) {
                        val body = resp.body()!!
                        
                        // Format code with space for readability (e.g. 123 456)
                        val formattedCode = if (body.code.length == 6) {
                            "${body.code.substring(0, 3)} ${body.code.substring(3)}"
                        } else {
                            body.code
                        }

                        txtCode.text = formattedCode
                        txtDetails.text = "Amount: $${body.amount}\nRef: ${body.reference.take(8)}..."
                        
                        layoutResult.visibility = View.VISIBLE
                    } else {
                        txtError.text = "Server Error: HTTP ${resp.code()}"
                        txtError.visibility = View.VISIBLE
                    }
                } catch (e: Exception) {
                    txtError.text = "Connection Failed: ${e.message}"
                    txtError.visibility = View.VISIBLE
                } finally {
                    btnGenerate.isEnabled = true
                    btnGenerate.text = "Generate Secure Code"
                }
            }
        }
    }
}
